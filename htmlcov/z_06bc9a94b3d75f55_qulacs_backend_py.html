1r"""The Qulacs backend. 

2 

3Qulacs is an optional dependency (`pip install eigenkit[qulacs]`). The package is 

4imported only when it is needed, so the rest of the library works without it, 

5and `compile()` raises a clear `ImportError` if it is missing. 

6 

7It declares `gradient_method = "adjoint"`. Qulacs provides an analytic 

8reverse-mode gradient (`ParametricQuantumCircuit.backprop`), so, as with the 

9NumPy reference simulator, the compiled circuit has a `jacobian` method, and the 

10gradient code of the core package prefers it to the parameter-shift rule, whose 

11cost is \(O(n_\mathrm{params})\). 

12 

13## Compile and run 

14 

15`compile()` builds one `ParametricQuantumCircuit` and the Qulacs `Observable` 

16objects once. Every rotation whose angle depends on the weights or inputs becomes 

17a *parametric slot*. `run()` sets the slots with `set_parameter`, without 

18rebuilding the circuit at each call, and `jacobian()` reads the gradients of all 

19slots from one `backprop` pass for each observable. 

20 

21## Conventions matched with the NumPy and PennyLane reference 

22 

23- **Rotation sign.** Qulacs rotations are \(e^{+i\theta P/2}\), while the library 

24 and PennyLane use \(e^{-i\theta P/2}\). Every rotation angle is negated when it 

25 is passed to Qulacs: the Qulacs parameter of a slot is 

26 \(\phi = -(\mathrm{coeff} \cdot \mathrm{angle})\). As a result, `backprop` 

27 returns \(d\langle O \rangle / d\phi\), and the chain rule through \(\phi\) gives 

28 \(d\langle O \rangle / d(\mathrm{weights})\) with the sign handled automatically 

29 (see `jacobian`). 

30- **Bit order.** Qulacs stores the amplitudes with qubit `0` as the least 

31 significant bit (as Qiskit does), while the library uses qubit `0` as the most 

32 significant bit. The statevector input and output, and the matrices of 

33 multi-qubit gates, are bit-reversed to match the two. The Pauli *observables* 

34 need no reindexing, because the qubit labels match: only the storage order of 

35 the amplitudes differs. 

36 

37## Gate coverage 

38 

39`add_parametric_R{X,Y,Z}` covers the trainable rotations directly. The gates that 

40have no parametric primitive in Qulacs are decomposed into these rotations, so 

41that they stay differentiable: 

42 

43- \(\mathrm{PhaseShift}(\theta)\) becomes a parametric `RZ`. The two are equal up 

44 to a global phase that cannot be observed. 

45- \(\mathrm{CRY}(\theta)\) becomes 

46 \(\mathrm{RY}(\theta/2)\, \mathrm{CX}\, \mathrm{RY}(-\theta/2)\, \mathrm{CX}\), where 

47 `CX` is a controlled-NOT gate. This decomposition is exact. 

48- \(\mathrm{CPHASE}(\theta)\) becomes 

49 \(P(\theta/2)_c\, \mathrm{CX}\, P(-\theta/2)_t\, \mathrm{CX}\, P(\theta/2)_t\), with 

50 each \(P\) as an `RZ` slot. This decomposition is exact up to a global phase 

51 that cannot be observed. 

52 

53`PhaseShift`, `CPHASE` and `CRY` with a constant angle, and the non-parametric 

54gates (`H`, `X`, `Z`, `S`, `Sdg`, `CNOT`, `SWAP`, `MCX`, `MCZ`, `QubitUnitary`, 

55`ControlledQubitUnitary`, `AmplitudeEncoding`), are added once as exact Qulacs 

56gates. `backprop` accepts these fixed gates, so native adjoint gradients are 

57available for every circuit. 

58 

59Example: 

60 ```python 

61 from eigenkit import Circuit, Encoding, Variational, all_z 

62 

63 blocks = [Encoding(in_features=4), Variational(depth=2)] 

64 

65 # by registered name 

66 circuit = Circuit(blocks=blocks, observables=all_z(4), provider="qulacs") 

67 

68 # or with the config, when a device setting matters 

69 from eigenkit.configs import QulacsConfig 

70 

71 circuit = Circuit( 

72 blocks=blocks, 

73 observables=all_z(4), 

74 provider=QulacsConfig(), 

75 ) 

76 ``` 

77 

78 Gradients are computed natively in reverse mode, with `backprop`. This backend 

79 needs the `qulacs` extra. 

80"""

462 r"""Compute the gradient with the native Qulacs method, the analytic adjoint. 

463 

464 The gradient code of the core package looks for a `jacobian` method. When 

465 a backend has one, it is used in preference to the parameter-shift rule, 

466 which works on any backend and is the fallback. Here the gradient comes 

467 from `backprop`: one pass for each observable gives 

468 \(d\langle O \rangle / d\phi\) for every parametric slot. These gradients are 

469 mapped onto the weights (and the inputs) with the same symbolic chain-rule 

470 factors as the parameter-shift code, so angles built with `Scale`, 

471 `Multiply`, `Sum` or `Negate` (for example in 

472 `Reuploading(trainable_scale=True)`) are also handled. 

473 

474 For a single sample, the two jacobians have shapes `(n_obs, n_weights)` 

475 and `(n_obs, n_inputs)`. For a 2-D `inputs` batch, they have a leading 

476 batch axis, and the batch is evaluated in a loop. 

477 

478 Args: 

479 weights (np.ndarray): The parameter vector (1-D). 

480 inputs (np.ndarray): The input features: 1-D for one sample, or 2-D 

481 with a leading batch axis. 

482 need_input_grad (bool): If True, also compute the jacobian with 

483 respect to the inputs. Default: `False`. 

484 

485 Returns: 

486 weight_jacobian: The jacobian with respect to the weights. 

487 input_jacobian: The jacobian with respect to the inputs, or `None` 

488 when `need_input_grad` is `False`. 

489 """

523 gate_grad = slot_grads[k] 

524 for w_index in collect_weight_dependencies(expr): 

525 factor = chain_factor(expr, w_index, weights, inputs) 

526 weight_jac[obs_index, w_index] += -coeff * factor * gate_grad 

527 if input_jac is not None: 

528 for x_index in collect_input_dependencies(expr): 

529 factor = chain_factor_input(expr, x_index, weights, inputs) 

530 input_jac[obs_index, x_index] += -coeff * factor * gate_grad 

531 return weight_jac, input_jac

535 def zero_state_probability(self, weights: np.ndarray, inputs: np.ndarray) -> float: 

536 r"""Return \(\lvert\langle 0 \cdots 0|\psi\rangle\rvert^2\), the squared 

537 amplitude of \(|0 \cdots 0\rangle\). 

538 

539 The basis state `0` is \(|0 \cdots 0\rangle\) whether qubit `0` is the least 

540 or the most significant bit, so no reindexing is needed. The fidelity 

541 quantum kernel is built on this method. 

542 

543 Args: 

544 weights (np.ndarray): The parameter vector (1-D). 

545 inputs (np.ndarray): The input features of one sample (1-D). 

546 

547 Returns: 

548 The probability, a float in \([0, 1]\). 

549 """ 

550 state = self._run_state( 

551 np.asarray(weights, dtype=np.float64), 

552 np.asarray(inputs, dtype=np.float64), 

553 ) 

554 amplitude = state.get_vector()[0] 

555 return float((amplitude.conjugate() * amplitude).real)

557 def statevector( 

558 self, weights: np.ndarray, inputs: np.ndarray, initial_state=None 

559 ) -> np.ndarray: 

560 r"""Return the full complex statevector after the circuit has run. 

561 

562 Qubit `0` is the most significant bit. `initial_state`, in the same 

563 order, is the starting state instead of \(|0 \cdots 0\rangle\). A 2-D 

564 `inputs` batch is evaluated in a loop and returns an array of shape 

565 `(B, 2**n)`. 

566 `initial_state` can then be one state of shape `(2**n,)`, shared by all 

567 samples, or a stack of shape `(B, 2**n)` with one state for each sample. 

568 

569 Args: 

570 weights (np.ndarray): The parameter vector (1-D). 

571 inputs (np.ndarray): The input features: 1-D for one sample, or 2-D 

572 with a leading batch axis. 

573 initial_state (np.ndarray | None): The state to start from, instead 

574 of \(|0 \cdots 0\rangle\). Default: `None`. 

575 

576 Returns: 

577 The statevector, of shape `(2**n_qubits,)`, or `(B, 2**n_qubits)` for 

578 a batch. 

579 """ 

580 weights = np.asarray(weights, dtype=np.float64) 

581 inputs = np.asarray(inputs, dtype=np.float64) 

582 if weights.ndim == 1 and inputs.ndim == 2: 

583 batch = inputs.shape[0] 

584 init = ( 

585 None 

586 if initial_state is None 

587 else validate_initial_state(initial_state, self._n_qubits, batch=batch) 

588 ) 

589 return np.stack( 

590 [ 

591 self._statevector_one(weights, row, None if init is None else init[i]) 

592 for i, row in enumerate(inputs) 

593 ] 

594 ) 

595 init = ( 

596 None 

597 if initial_state is None 

598 else validate_initial_state(initial_state, self._n_qubits) 

599 ) 

600 return self._statevector_one(weights, inputs, init) 

601 

602 def _statevector_one(self, weights, inputs, init_flat) -> np.ndarray: 

603 state = self._run_state(weights, inputs, initial_flat=init_flat) 

604 vector = _swap_endianness( 

605 np.asarray(state.get_vector(), dtype=np.complex128), self._n_qubits 

606 ) 

607 return vector * self._phase_correction(weights, inputs) 

608 

609 def _phase_correction(self, weights, inputs) -> complex: 

610 """The factor that turns qulacs' state back into the circuit's own. 

611 

612 Each trainable phase gate is lowered to RZ slots, which agree with the 

613 gate up to `e^{-i·c·theta}`. Multiplying by its inverse here makes 

614 `statevector()` return the same amplitudes as every other backend -- 

615 phases included, which matters wherever the state itself is the output 

616 rather than an expectation value. 

617 """ 

618 if not self._phase_terms: 

619 return 1.0 + 0.0j 

620 total = sum( 

621 coefficient * float(resolve(expression, weights, inputs)) 

622 for expression, coefficient in self._phase_terms 

623 ) 

624 return complex(np.exp(1j * total))

627class QulacsBackend: 

628 r"""The provider for the Qulacs statevector simulator. 

629 

630 It is optional: it is registered as `"qulacs"` when the package is installed, 

631 and the package is imported only on first use. It declares 

632 `gradient_method = "adjoint"`, so the gradient code of the core package uses 

633 [`QulacsCompiled.jacobian()`][eigenkit.core.backends.qulacs_backend.QulacsCompiled.jacobian] 

634 (analytic, in reverse mode, through Qulacs `backprop`) instead of the 

635 parameter-shift rule, whose cost is \(O(n_\mathrm{params})\). 

636 """ 

637 

638 gradient_method = "adjoint" 

639 

640 def __init__( 

641 self, 

642 config: QulacsConfig | None = None, 

643 name: str = "qulacs", 

644 ): 

645 """Build the backend from a device configuration. 

646 

647 Args: 

648 config (QulacsConfig): The device settings. Default: `None`, which 

649 uses `QulacsConfig()`. 

650 name (str): The registry name, used in error messages. Default: 

651 `"qulacs"`. 

652 """ 

653 self.name = name 

654 self.config = config or QulacsConfig() 

655 

656 @property 

657 def capabilities(self) -> Capabilities: 

658 """What this backend supports. 

659 

660 It runs every gate, returns exact expectation values only, does not 

661 evaluate a batch in one call, and has no limit on the number of qubits. 

662 The adjoint of `AmplitudeEncoding` is excluded, because a state 

663 preparation cannot be inverted as a Qulacs gate. See 

664 [`Capabilities`][eigenkit.core.provider.Capabilities]. 

665 """ 

666 return Capabilities( 

667 supported_gates=ALL_GATES, 

668 supports_batching=False, 

669 supports_shots=False, 

670 max_qubits=None, 

671 # State preparation is not invertible as a qulacs gate, so the 

672 # adjoint of AmplitudeEncoding is rejected (mirrors Qiskit). 

673 supported_adjoint_gates=ALL_GATES - frozenset({"AmplitudeEncoding"}), 

674 ) 

675 

676 def compile( 

677 self, 

678 circuit: CircuitSpec, 

679 observables: Sequence[Observable], 

680 ) -> QulacsCompiled: 

681 """Prepare `circuit` for repeated evaluation of `observables`. 

682 

683 A `Circuit` calls it once and reuses the result for every call, so the 

684 expensive work belongs here. 

685 

686 Args: 

687 circuit (CircuitSpec): The circuit to prepare. 

688 observables (Sequence[Observable]): The observables to measure. 

689 

690 Returns: 

691 The prepared circuit, as a `QulacsCompiled` object. 

692 """ 

693 return QulacsCompiled(circuit, observables, self.config)
