1"""Warn about ansatz settings whose weights or data cannot reach the output. 

2 

3Every layer here builds a circuit of the same form: a rotation encoding, a 

4variational block of rotations followed by a controlled-NOT (CNOT) ladder, and a 

5Pauli measurement on each qubit. Some combinations of these three settings make 

6the circuit degenerate, which can be proved: either the variational weights 

7cancel exactly out of the expectation value, or the input does. Such a layer 

8runs, returns plausible numbers and trains the classical layers around it, so 

9nothing later in the model shows that the circuit has no effect. 

10 

11Four separate mechanisms cause this, and this module tells them apart instead of 

12grouping them under a rule to "keep the axes distinct". That rule is neither 

13necessary nor sufficient. `RY`/`RY`/`Y` shares an axis and works on two qubits or 

14more, because the CNOT ladder does not preserve `Y`-strings. A single `RZ` 

15encoding has no effect whatever the other two axes are. 

16 

17 1. An `RZ` encoding with a single encoding layer. The circuit starts in 

18 `|0...0>`, which is an eigenstate of every `RZ`, so the encoding only adds 

19 a phase. Layers that encode again between variational blocks are not 

20 affected, because their second encoding acts on a state that has already 

21 changed. 

22 

23 2. The variational block commutes with the observable. `RZ` commutes with 

24 `Z`-strings and `RX` with `X`-strings, and the CNOT ladder maps `Z`-strings 

25 to `Z`-strings and `X`-strings to `X`-strings. So with 

26 `variational_type="basic"` and the rotation on the measurement axis, the 

27 weights cancel exactly. CNOT does *not* preserve `Y`-strings 

28 (`YI -> YX`, `IY -> ZY`), which is why the `Y` case is degenerate only on 

29 a single qubit, where there is no ladder. 

30 

31 3. A circuit with only real gates, measured in `Y`. `RY` and CNOT are real 

32 matrices, so an `RY` encoding followed by `RY` rotations keeps the 

33 statevector real, and `<Y>` is exactly zero for a real state. 

34 

35 4. A circuit with only diagonal gates. An `RZ` encoding with `RZ` rotations 

36 never leaves `|0...0>`, whatever the measurement basis is. 

37 

38The rule in `degenerate_reason` below was found by trying every layer, number of 

39qubits, depth, rotation, basis and variational type, and comparing the result 

40with the measured gradients. 

41`tests/layers/test_degenerate_ansatz_warning.py` runs this search again: the 

42warning must be raised exactly when a gradient or the sensitivity to the data is 

43zero, so that neither a false warning nor a degenerate layer without a warning 

44can go unnoticed. 

45""" 

46 

47from __future__ import annotations 

48 

49import warnings 

50 

51 

52_AXIS = {"RX": "X", "RY": "Y", "RZ": "Z"} 

53 

54# CNOT conjugation maps X-strings to X-strings and Z-strings to Z-strings, but 

55# sends Y-strings to mixed ones. So a shared axis only cancels the weights for 

56# X and Z — unless there is no ladder at all. 

57_LADDER_PRESERVES = ("X", "Z") 

58 

59 

60def degenerate_reason( 

61 *, 

62 enc_rotation: str, 

63 var_rotation: str, 

64 variational_type: str, 

65 basis: str, 

66 n_qubits: int, 

67 n_encodings: int, 

68) -> str | None: 

69 """Return why this configuration is degenerate, or `None` if it is not. 

70 

71 Args: 

72 enc_rotation: The encoding rotation: `"RX"`, `"RY"` or `"RZ"`. 

73 var_rotation: The variational rotation: `"RX"`, `"RY"` or `"RZ"`. It is 

74 used only when `variational_type` is `"basic"`. 

75 variational_type: `"basic"` or `"strong"`. 

76 basis: The measurement basis: `"X"`, `"Y"` or `"Z"`. 

77 n_qubits: The number of qubits in one circuit. `1` means that there is no 

78 controlled-NOT (CNOT) ladder. 

79 n_encodings: The number of times the input is encoded into one circuit. 

80 It is `1` for a single encoding and `depth` for reuploading. 

81 

82 Returns: 

83 A sentence that names the mechanism and how to avoid it, or `None`. It is 

84 also `None` when either rotation names no axis. The layers call this 

85 function before they build their blocks, and rejecting a wrong 

86 rotation is the job of the block. Reading `_AXIS` without this check 

87 would replace the `ValueError` of the block with a `KeyError` from a 

88 private module, and checking only the rotation that was valid would 

89 warn about a configuration that is rejected anyway. 

90 """ 

91 enc_axis = _AXIS.get(enc_rotation) 

92 var_axis = _AXIS.get(var_rotation) 

93 if enc_axis is None or var_axis is None: 

94 return None 

95 is_basic = variational_type == "basic" 

96 

97 if enc_axis == "Z" and var_axis == "Z" and is_basic: 

98 return ( 

99 "every gate is diagonal (RZ encoding, RZ rotations, CNOT ladder), so " 

100 "the state never leaves |0...0> and the output is a constant. Encode " 

101 'on "RX" or "RY", and rotate off the measurement axis.' 

102 ) 

103 

104 if enc_axis == "Z" and n_encodings == 1: 

105 return ( 

106 "the circuit starts in |0...0>, which every RZ leaves alone up to a " 

107 'phase, so the input never reaches the measurement. Use enc_rotation="RX" ' 

108 'or "RY".' 

109 ) 

110 

111 if is_basic and var_axis == basis: 

112 ladder_cancels = basis in _LADDER_PRESERVES or n_qubits == 1 

113 encoding_rescues = n_encodings > 1 and enc_axis != basis 

114 if ladder_cancels and not encoding_rescues: 

115 return ( 

116 f"the variational block commutes with the <{basis}> observable " 

117 f"({var_rotation} rotations and the CNOT ladder both preserve " 

118 f"{basis}-strings), so the weights cancel out of the expectation " 

119 "value and every gradient is exactly zero. Rotate off the " 

120 'measurement axis, or use variational_type="strong".' 

121 ) 

122 

123 if is_basic and enc_axis == "Y" and var_axis == "Y" and basis == "Y": 

124 return ( 

125 "RY and CNOT are real matrices, so the statevector stays real and " 

126 "<Y> is identically zero on it. Measure in another basis, or move the " 

127 "encoding or the rotation off the Y axis." 

128 ) 

129 

130 return None 

131 

132 

133def warn_if_degenerate( 

134 *, 

135 layer_name: str, 

136 enc_rotation: str, 

137 var_rotation: str, 

138 variational_type: str, 

139 basis: str, 

140 n_qubits: int, 

141 n_encodings: int, 

142 stacklevel: int = 3, 

143) -> None: 

144 """Issue a `UserWarning` if this configuration cannot train. 

145 

146 The layer is still built. The caller decides which circuit to run, and there 

147 are valid reasons to build a degenerate one on purpose: it is exactly the 

148 baseline for checking whether the quantum part contributes anything. 

149 """ 

150 reason = degenerate_reason( 

151 enc_rotation=enc_rotation, 

152 var_rotation=var_rotation, 

153 variational_type=variational_type, 

154 basis=basis, 

155 n_qubits=n_qubits, 

156 n_encodings=n_encodings, 

157 ) 

158 if reason is None: 

159 return 

160 warnings.warn( 

161 f"{layer_name}(enc_rotation={enc_rotation!r}, var_rotation={var_rotation!r}, " 

162 f"variational_type={variational_type!r}, basis={basis!r}) is degenerate: " 

163 f"{reason}", 

164 UserWarning, 

165 stacklevel=stacklevel, 

166 )
