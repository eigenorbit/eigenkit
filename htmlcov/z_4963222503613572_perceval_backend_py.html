1"""The Perceval executor for photonic circuits (experimental). 

2 

3It translates a 

4[`PhotonicCircuitSpec`][eigenkit.photonics.ir.PhotonicCircuitSpec] into a 

5Perceval `Circuit`, and computes the output distribution of photon numbers with an 

6exact Perceval simulator, by default simulation of linear optical systems (SLOS). 

7It is the linear-optics counterpart of a qubit `Provider`, but it does **not** 

8implement the qubit `Provider` protocol: there is no `CircuitSpec`, no Pauli 

9`Observable` and no array of expectation values. This is why photonics is a 

10separate package. 

11 

12`perceval` is imported only *inside* the methods, when they run, so importing this 

13module (and therefore `import eigenkit`) never imports the Perceval software 

14development kit (SDK). The data module `photonics.ir` never uses Perceval at all. 

15 

16The work is split between compilation and evaluation, as in the qubit providers: 

17`compile()` is called once, and `run(weights, inputs)` at every call. For now, the 

18Perceval circuit is built again at every call, after the `Parameter` expressions 

19are evaluated to numbers. Building it once with symbolic `pcvl.P` parameters, and 

20setting their values at each call, is not done yet. (This is the saving that the 

21qubit providers get from the split between compilation and evaluation.) 

22"""

71def build_perceval_circuit(spec, weights, inputs): 

72 """Translate a `PhotonicCircuitSpec` into a Perceval `Circuit`. 

73 

74 The `Parameter` angles are evaluated to numbers first. The local and the 

75 remote executors both use this function, so they build exactly the same 

76 circuit. (The live test of the remote executor compares sampled and exact 

77 results on the *same* circuit.) 

78 

79 Args: 

80 spec (PhotonicCircuitSpec): The circuit to translate. 

81 weights (np.ndarray): The 1-D weight vector used to evaluate the angles. 

82 inputs (np.ndarray): The 1-D input vector used to evaluate the angles. 

83 

84 Returns: 

85 circuit (perceval.Circuit): The circuit, on `spec.n_modes` modes. 

86 

87 Raises: 

88 ValueError: If `weights` or `inputs` is not 1-D with the length that the 

89 spec declares. 

90 """ 

91 import perceval as pcvl 

92 

93 _check_lengths(spec, weights, inputs) 

94 circ = pcvl.Circuit(spec.n_modes) 

95 for comp in spec.components: 

96 if isinstance(comp, PhaseShifter): 

97 circ.add(comp.mode, pcvl.PS(float(resolve(comp.phi, weights, inputs)))) 

98 elif isinstance(comp, BeamSplitter): 98 ↛ 104line 98 didn't jump to line 104 because the condition on line 98 was always true

99 theta = float(resolve(comp.theta, weights, inputs)) 

100 phi = float(resolve(comp.phi, weights, inputs)) 

101 # Our single `phi` maps to perceval's top-right phase; the exact BS unitary 

102 # (Rx convention, R = cos^2(theta/2)) is pinned by the parity tests. 

103 circ.add(tuple(comp.modes), pcvl.BS(theta=theta, phi_tr=phi)) 

104 elif isinstance(comp, Permutation): 

105 circ.add(0, pcvl.PERM(list(comp.perm))) 

106 else: # pragma: no cover - guarded by the Component union 

107 raise TypeError(f"unknown photonic component: {type(comp).__name__}") 

108 return circ

111def fock_tuple(state, n_modes: int) -> tuple[int, ...]: 

112 """Convert a Perceval `BasicState`, or any indexable object, to a tuple of photon counts. 

113 

114 Args: 

115 state (BasicState | Sequence[int]): The number of photons in each mode, 

116 indexed by mode. 

117 n_modes (int): The number of modes to read. 

118 

119 Returns: 

120 The number of photons in each mode, as a tuple of integers. 

121 """ 

122 return tuple(int(state[i]) for i in range(n_modes))

126 """A reusable handle to a compiled photonic circuit, cheap to call many times.

128 It is the linear-optics counterpart of `CompiledCircuit`. 

129 `PercevalBackend.compile` builds it.

131 Args: 

132 spec (PhotonicCircuitSpec): The circuit that this handle evaluates. 

133 backend_name (str): The Perceval backend, as for `PercevalBackend`. 

134 Default: `"SLOS"`. 

135 

136 Raises: 

137 ValueError: If `backend_name` is unknown or only samples.

142 self._spec = spec 

143 self._backend_name = backend_name

145 def _build_circuit(self, weights, inputs): 

146 return build_perceval_circuit(self._spec, weights, inputs)

148 def run(self, weights, inputs) -> FockDistribution: 

149 """Return the output distribution of photon numbers. 

150 

151 The distribution is a dictionary `{photon_counts_tuple: probability}`. Fock 

152 states with probability zero may be left out, because Perceval's 

153 `prob_distribution` is sparse. 

154 

155 Args: 

156 weights (np.ndarray): The 1-D weight vector, of length `spec.n_weights`. 

157 inputs (np.ndarray): The 1-D input vector, of length `spec.n_inputs`. 

158 

159 Raises: 

160 ValueError: If `weights` or `inputs` has the wrong shape. 

161 """ 

162 import perceval as pcvl 

163 

164 circ = self._build_circuit(weights, inputs) 

165 backend = pcvl.BackendFactory.get_backend(self._backend_name) 

166 backend.set_circuit(circ) 

167 backend.set_input_state(pcvl.BasicState(list(self._spec.input_state))) 

168 dist = backend.prob_distribution() 

169 return { 

170 fock_tuple(state, self._spec.n_modes): float(prob) 

171 for state, prob in dist.items() 

172 }

242class PercevalBackend: 

243 """The photonic executor that runs on a local Perceval simulator. 

244 

245 `compile(spec)` returns a `PercevalCompiled`. 

246 

247 Args: 

248 backend_name (str): A Perceval backend that computes the full output 

249 distribution. `"SLOS"` (the default) and `"Naive"` are exact, while 

250 `"MPS"` and `"NaiveApprox"` approximate the distribution, and so the 

251 gradients too. Backends that only sample, such as 

252 `"CliffordClifford2017"`, are rejected, because `run` returns 

253 probabilities, not samples. 

254 

255 Raises: 

256 ValueError: If `backend_name` is unknown or only samples. 

257 """ 

258 

259 def __init__(self, backend_name: str = "SLOS") -> None: 

260 _check_backend_name(backend_name) 

261 self.backend_name = backend_name 

262 

263 def compile(self, spec: PhotonicCircuitSpec) -> PercevalCompiled: 

264 """Wrap `spec` in a reusable handle. Perceval runs at each call of the handle. 

265 

266 Args: 

267 spec (PhotonicCircuitSpec): The circuit to compile. 

268 

269 Returns: 

270 The `PercevalCompiled` handle for `spec`. 

271 """ 

272 return PercevalCompiled(spec, backend_name=self.backend_name)
