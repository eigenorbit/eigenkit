15def apply_hartree_fock(circuit: CircuitBuilder, n_electrons: int) -> CircuitBuilder: 

16 r"""Prepare the Hartree-Fock reference state on an existing builder. 

17 

18 It applies an `X` gate to each of the lowest `n_electrons` spin-orbitals, which 

19 marks them as occupied (\(|1\rangle\)). 

20 

21 The convention is the one of the Jordan-Wigner mapping: mode `j` is qubit `j`, 

22 and the spin-orbitals are ordered so that the lowest `n_electrons` of them are 

23 occupied in the reference state, for example \(|1 \cdots 1\, 0 \cdots 0\rangle\). 

24 

25 Args: 

26 circuit (CircuitBuilder): The builder to add the gates to. 

27 n_electrons (int): The number of electrons, which sets the Hartree-Fock 

28 reference state. 

29 

30 Returns: 

31 The same builder, so that calls can be chained. 

32 """ 

33 for qubit in range(n_electrons): 

34 circuit.x(qubit) 

35 return circuit

38def hartree_fock_state(n_qubits: int, n_electrons: int) -> CircuitSpec: 

39 r"""A circuit that prepares the Hartree-Fock reference state. 

40 

41 The state is \(|1 \cdots 1\, 0 \cdots 0\rangle\) on `n_qubits` qubits, with 

42 `n_electrons` occupied spin-orbitals. 

43 

44 Args: 

45 n_qubits (int): The number of qubits. 

46 n_electrons (int): The number of electrons, which sets the Hartree-Fock 

47 reference state. 

48 

49 Returns: 

50 The circuit, as a `CircuitSpec` with no trainable weights. 

51 """ 

52 if not 0 <= n_electrons <= n_qubits: 

53 raise ValueError( 

54 f"n_electrons ({n_electrons}) must be between 0 and n_qubits ({n_qubits})" 

55 ) 

56 builder = CircuitBuilder(n_qubits=n_qubits) 

57 apply_hartree_fock(builder, n_electrons) 

58 return builder.build()

61def _single_excitation_generator(occupied: int, virtual: int) -> FermionOperator: 

62 r"""The anti-Hermitian single-excitation generator \(a_v^\dagger a_o - a_o^\dagger a_v\).""" 

63 excitation = FermionOperator.creation(virtual) * FermionOperator.annihilation( 

64 occupied 

65 ) 

66 return excitation - excitation.hermitian_conjugate()

70 occ_i: int, occ_j: int, virt_a: int, virt_b: int 

71) -> FermionOperator: 

72 r"""The anti-Hermitian double-excitation generator 

73 \(a_a^\dagger a_b^\dagger a_j a_i - \text{h.c.}\).""" 

74 excitation = ( 

75 FermionOperator.creation(virt_a) 

76 * FermionOperator.creation(virt_b) 

77 * FermionOperator.annihilation(occ_j) 

78 * FermionOperator.annihilation(occ_i) 

79 ) 

80 return excitation - excitation.hermitian_conjugate()

83def uccsd_excitation_generators(n_qubits: int, n_electrons: int) -> list[FermionOperator]: 

84 r"""The excitation generators \(T - T^\dagger\) of the UCCSD ansatz. 

85 

86 UCCSD is the unitary coupled-cluster ansatz with single and double excitations. 

87 The excitations go from the occupied spin-orbitals (the lowest `n_electrons`) 

88 to the unoccupied ones. Single and double excitations are taken over all choices 

89 of occupied and unoccupied indices, with no restriction from spin symmetry, 

90 which keeps the construction simple and correct. A variant restricted by spin 

91 symmetry, which would have fewer parameters, is not implemented. 

92 

93 Args: 

94 n_qubits (int): The number of qubits. 

95 n_electrons (int): The number of electrons, which sets the Hartree-Fock 

96 reference state. 

97 

98 Returns: 

99 The generators, as a list of `FermionOperator` objects. 

100 """ 

101 occupied = list(range(n_electrons)) 

102 virtual = list(range(n_electrons, n_qubits)) 

103 generators: list[FermionOperator] = [] 

104 for occ in occupied: 

105 for virt in virtual: 

106 generators.append(_single_excitation_generator(occ, virt)) 

107 for occ_i, occ_j in combinations(occupied, 2): 

108 for virt_a, virt_b in combinations(virtual, 2): 

109 generators.append(_double_excitation_generator(occ_i, occ_j, virt_a, virt_b)) 

110 return generators

113def uccsd_ansatz( 

114 n_qubits: int, 

115 n_electrons: int, 

116 reps: int = 1, 

117) -> CircuitSpec: 

118 r"""The UCCSD ansatz on the Hartree-Fock reference, with a Trotter approximation. 

119 

120 UCCSD is the unitary coupled-cluster ansatz with single and double excitations. 

121 The circuit prepares 

122 

123 \[ 

124 \Big(\prod_k \exp\big(\theta_k (T_k - T_k^\dagger)\big)\Big)^{\mathrm{reps}} 

125 |\mathrm{HF}\rangle, 

126 \] 

127 

128 where \(|\mathrm{HF}\rangle\) is the Hartree-Fock state. 

129 

130 Each generator \(G = T - T^\dagger\) is anti-Hermitian. The Jordan-Wigner mapping 

131 turns \(H = -iG\) into a Hermitian qubit operator, and 

132 \(\exp(\theta G) = \exp(i\theta H)\) is then a product of `pauli_exponential` gates. 

133 The Pauli terms of one generator commute, so that product is exact. The product 

134 over different generators is a first-order Trotter approximation. `reps` repeats 

135 the whole sequence with independent weights, which reduces the Trotter error: 

136 more repetitions bring the circuit closer to the exact UCCSD unitary. The ansatz 

137 has `reps * n_excitations` weights in total. 

138 

139 Args: 

140 n_qubits (int): The number of qubits, one per spin-orbital. 

141 n_electrons (int): The number of electrons, which sets the Hartree-Fock 

142 reference and the possible excitations. 

143 reps (int): The number of Trotter repetitions of the product of 

144 excitations. The default is `1`. 

145 

146 Returns: 

147 The ansatz as a `CircuitSpec`, with one trainable angle per generator and 

148 per repetition. 

149 """ 

150 if reps < 1: 150 ↛ 151line 150 didn't jump to line 151 because the condition on line 150 was never true

151 raise ValueError(f"reps must be >= 1, got {reps}") 

152 if not 0 < n_electrons < n_qubits: 

153 # Outside this range there are no excitations, and the "ansatz" would 

154 # come back with zero weights: VQE would then have nothing to optimise, 

155 # converge immediately and report success on a circuit that cannot 

156 # represent anything. n_electrons > n_qubits would fail later and less 

157 # clearly, with a message about a qubit index being out of range. 

158 raise ValueError( 

159 f"n_electrons must be between 1 and n_qubits - 1 for UCCSD to have " 

160 f"any excitations; got n_electrons={n_electrons} with " 

161 f"n_qubits={n_qubits}. Zero electrons and a fully occupied shell " 

162 f"both leave nothing to excite." 

163 ) 

164 generators = uccsd_excitation_generators(n_qubits, n_electrons) 

165 # -1j * G is Hermitian; cache its real Pauli terms once. 

166 hermitian_terms = [jordan_wigner(-1j * g)[0].terms for g in generators] 

167 builder = CircuitBuilder(n_qubits=n_qubits) 

168 thetas = builder.add_weights(len(generators) * reps) 

169 apply_hartree_fock(builder, n_electrons) 

170 index = 0 

171 for _ in range(reps): 

172 for terms in hermitian_terms: 

173 theta = thetas[index] 

174 index += 1 

175 for term in terms: 

176 # exp(i theta c_k P_k) = pauli_exponential(P_k, theta, -c_k). 

177 builder.pauli_exponential(term.paulis, theta, coefficient=-term.coeff) 

178 return builder.build()
