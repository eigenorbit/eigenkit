1r"""Fermion-to-qubit mappings: build a qubit `Observable` from a 

2`FermionOperator`. 

3 

4Jordan-Wigner convention (OpenFermion-compatible): 

5 

6\[ 

7a_j = \Big(\prod_{k<j} Z_k\Big)\,\frac{X_j + i Y_j}{2} 

8\qquad 

9a_j^\dagger = \Big(\prod_{k<j} Z_k\Big)\,\frac{X_j - i Y_j}{2} 

10\] 

11 

12so the number operator maps to \(a_j^\dagger a_j = (I - Z_j)/2\), with an 

13occupied orbital being qubit \(|1\rangle\). The parity string 

14\(\prod_{k<j} Z_k\) is what enforces fermionic antisymmetry. 

15 

16`jordan_wigner` returns ``(Observable, offset)``: the constant (identity) 

17term is returned separately as ``offset`` because the Pauli backends reject 

18identity-only terms, and a variational solver adds it back to \(\langle H \rangle\). 

19""" 

20 

21from __future__ import annotations

23from ._pauli import PauliSum 

24from .fermion import FermionOperator

27def _jordan_wigner_ladder(mode: int, action: int) -> PauliSum: 

28 """JW image of a single ladder operator on ``mode``. 

29 

30 ``action = 1`` (creation) uses ``(X - iY)/2``; ``action = 0`` (annihilation) 

31 uses ``(X + iY)/2``; both carry the parity string ``Z_0 ... Z_{mode-1}``. 

32 """ 

33 parity = {k: "Z" for k in range(mode)} 

34 x_string = dict(parity) 

35 x_string[mode] = "X" 

36 y_string = dict(parity) 

37 y_string[mode] = "Y" 

38 

39 y_sign = -1j if action == 1 else 1j # creation: -iY, annihilation: +iY 

40 result = PauliSum() 

41 result._add_term(tuple(sorted(x_string.items())), 0.5) 

42 result._add_term(tuple(sorted(y_string.items())), 0.5 * y_sign) 

43 return result

46def jordan_wigner(operator: FermionOperator, *, atol: float = 1e-10): 

47 """Map a `FermionOperator` to qubits with the Jordan-Wigner transformation. 

48 

49 The operator must be Hermitian, as a Hamiltonian is. If a resulting 

50 coefficient is not real, a `ValueError` is raised. 

51 

52 Args: 

53 operator (FermionOperator): The fermionic operator to map. 

54 atol (float): Terms whose coefficient is smaller than this value in 

55 magnitude are dropped. 

56 

57 Returns: 

58 observable (Observable): The operator on qubits, without its identity part. 

59 offset (float): The coefficient of the identity part, which the 

60 `Observable` does not contain. 

61 """ 

62 total = PauliSum() 

63 for term, coefficient in operator.terms.items(): 

64 product = PauliSum.identity(coefficient) 

65 for mode, action in term: 

66 product = product * _jordan_wigner_ladder(mode, action) 

67 total = total + product 

68 return total.to_observable(atol=atol)

77 mode index plus one. 

78 

79 Args: 

80 operator (FermionOperator): The fermionic operator to map. 

81 n_qubits (int): The number of qubits. 

82 

83 Returns: 

84 observable (Observable): The operator on qubits, without its identity part. 

85 offset (float): The coefficient of the identity part, which the 

86 `Observable` does not contain. 

87 """
