1r"""VQE — the Variational Quantum Eigensolver, on the core spine. 

2 

3Given a qubit Hamiltonian (an `Observable` plus its constant offset), VQE 

4minimizes \(\langle H \rangle\) over a parametrized ansatz to estimate the ground-state 

5energy. It is a thin layer over `minimize_expectation`: build an ansatz, wrap 

6it in a numpy `Circuit` whose observable is the Hamiltonian, optimize. Pure 

7core (numpy/scipy), torch-free, backend-agnostic. 

8 

9The default ansatz is a hardware-efficient one (RY/RZ rotations + a linear 

10CNOT entangler) starting from \(|0 \cdots 0\rangle\), so it needs no Hartree-Fock 

11preparation. For a chemistry-specific ansatz, pass `uccsd_ansatz(...)` as 

12`ansatz`; it starts from the Hartree-Fock reference itself. 

13"""

29def hardware_efficient_ansatz(n_qubits: int, layers: int) -> CircuitSpec: 

30 """A standard hardware-efficient ansatz. 

31 

32 It starts with a layer of `RY` rotations, then repeats `layers` times a linear 

33 chain of `CNOT` gates followed by an `RY` and an `RZ` rotation on each qubit. It 

34 has `n_qubits + 2 * layers * n_qubits` trainable weights. 

35 

36 Args: 

37 n_qubits (int): The number of qubits. 

38 layers (int): The number of repetitions. 

39 

40 Returns: 

41 The ansatz, as a `CircuitSpec`. 

42 """ 

43 qc = CircuitBuilder(n_qubits=n_qubits) 

44 weights = qc.add_weights(n_qubits + 2 * layers * n_qubits) 

45 index = 0 

46 for qubit in range(n_qubits): 

47 qc.ry(weights[index], qubit) 

48 index += 1 

49 for _ in range(layers): 

50 for qubit in range(n_qubits - 1): 

51 qc.cnot(qubit, qubit + 1) 

52 for qubit in range(n_qubits): 

53 qc.ry(weights[index], qubit) 

54 index += 1 

55 qc.rz(weights[index], qubit) 

56 index += 1 

57 return qc.build()

60@dataclass(frozen=True, eq=False) 

61class VQEResult(ValueEquality): 

62 """Outcome of a run of the Variational Quantum Eigensolver (VQE). 

63 

64 The result is read-only. Assigning to a field raises `FrozenInstanceError`, and 

65 writing into an array field raises `ValueError`. This prevents a mistyped `=` 

66 (instead of `==`) or an in-place edit from leaving a result whose scalar fields 

67 describe a different run from the array it holds. To get a writable copy of the 

68 array, use `np.array(result.field)`. 

69 

70 Results compare by value, including their arrays. They are not hashable, 

71 because a result is a collection of numbers and not a key, and a correct hash 

72 of a float array is expensive to compute. 

73 

74 Attributes: 

75 energy: The lowest energy found, with the offset added back. For a 

76 molecule, it is in Hartree. 

77 optimal_parameters: The ansatz weights that produced this energy. 

78 n_iterations: The number of optimiser iterations in the best restart. It 

79 is not summed over restarts. 

80 success: Whether the optimiser of that restart reported convergence. 

81 """ 

82 

83 energy: float 

84 optimal_parameters: np.ndarray 

85 n_iterations: int 

86 success: bool 

87 

88 def __post_init__(self) -> None: 

89 """Make the array field read-only. 

90 

91 Freezing the dataclass prevents a field from being reassigned, but it does not 

92 prevent writing into the array that the field holds, and a result that has been 

93 written into still reports the scalar values of the original run. The result 

94 therefore keeps its own copy, so the array it was built from can still be 

95 changed by the caller. To get a writable copy, use 

96 `np.array(result.optimal_parameters)`. 

97 """ 

98 freeze_arrays(self, "optimal_parameters")

101class VQE(BaseQuantumSolver): 

102 """Variational Quantum Eigensolver (VQE). 

103 

104 Args: 

105 layers: The depth of the default hardware-efficient ansatz. The default is 

106 `2`. 

107 provider: The backend to use. The default is `"numpy"`, the built-in 

108 reference simulator. 

109 optimizer: The optimiser: the name of a `scipy.optimize.minimize` method, 

110 `"spsa"`, or a custom callable (see `minimize_expectation`). The 

111 default is `"L-BFGS-B"`. 

112 restarts: The number of runs from random starting points. The lowest 

113 energy is kept, because variational landscapes have local minima. The 

114 default is `1`. 

115 random_state: The base seed. Restart `r` uses `random_state + r`, and 

116 `None` leaves every restart unseeded. The default is `42`. 

117 """ 

118 

119 def __init__( 

120 self, 

121 layers: int = 2, 

122 *, 

123 provider: ProviderSpec = "numpy", 

124 optimizer: Optimizer = "L-BFGS-B", 

125 restarts: int = 1, 

126 random_state: int | None = 42, 

127 ): 

128 if restarts < 1: 

129 raise ValueError(f"restarts must be >= 1, got {restarts}") 

130 super().__init__(provider=provider) 

131 self.layers = layers 

132 self.optimizer = optimizer 

133 self.restarts = restarts 

134 self.random_state = random_state 

135 

136 def solve( 

137 self, 

138 hamiltonian: Observable, 

139 n_qubits: int, 

140 *, 

141 offset: float = 0.0, 

142 ansatz: CircuitSpec | None = None, 

143 ) -> VQEResult: 

144 """Minimise the expectation value of the Hamiltonian over the ansatz. 

145 

146 It runs `restarts` independent optimisations from different random starting 

147 points and keeps the lowest energy, because a variational landscape has local 

148 minima. 

149 

150 Args: 

151 hamiltonian: The Hamiltonian to minimise, as a qubit `Observable`, usually 

152 the one returned by `jordan_wigner`. 

153 n_qubits: The number of qubits of the circuit. It must match the 

154 Hamiltonian. 

155 offset: The constant added to the minimised value. This is where the 

156 identity term returned by the mapping goes, which for a molecule 

157 includes the nuclear repulsion, in Hartree. If it is left out, the 

158 energy is wrong by exactly that constant. The default is `0.0`. 

159 ansatz: The circuit to optimise. The default, `None`, uses the built-in 

160 hardware-efficient ansatz with `layers` layers. 

161 

162 Returns: 

163 A `VQEResult` with the best energy, its parameters, and whether that 

164 restart converged. 

165 """ 

166 spec = ( 

167 ansatz 

168 if ansatz is not None 

169 else hardware_efficient_ansatz(n_qubits, self.layers) 

170 ) 

171 circuit = Circuit( 

172 spec, observables=hamiltonian, provider=self.provider, interface="numpy" 

173 ) 

174 best = None 

175 for restart in range(self.restarts): 

176 result = minimize_expectation( 

177 circuit, 

178 optimizer=self.optimizer, 

179 random_state=( 

180 None if self.random_state is None else self.random_state + restart 

181 ), 

182 ) 

183 if best is None or result.optimal_value < best.optimal_value: 

184 best = result 

185 return VQEResult( 

186 energy=best.optimal_value + offset, 

187 optimal_parameters=best.optimal_parameters, 

188 n_iterations=best.n_iterations, 

189 success=best.success, 

190 )
