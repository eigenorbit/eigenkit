1"""The core `Circuit`, which needs no framework, compiles once and accepts batches. 

2 

3- The circuit is compiled once, in `__init__`, and every call (`__call__`) reuses 

4 the compiled handle. 

5- `inputs` can be 1-D (one sample) or 2-D (a batch, with the samples along the 

6 first axis). The array is always passed with its shape unchanged to 

7 `compiled.run` or `compiled.run_torch`, and the backend handles the shape 

8 itself. Backends that cannot run a batch at once use `batch_call`. 

9- The observables can be given in several forms. `None` measures `Z` on every 

10 qubit. Otherwise, pass an `Observable`, a list of them, or a `PauliExpr`. 

11 

12This module holds the NumPy path: forward values computed with NumPy only, and 

13analytic jacobians from the parameter-shift rule or the chain rule. Automatic 

14differentiation in a framework (PyTorch now, JAX later) is handled by a 

15registered *interface* (see `interfaces.py`), so this module imports no framework 

16and stays in the lightweight core. The PyTorch `nn.Module` wrapper, `TorchCircuit`, 

17is in the PyTorch package (`eigenkit.torch`). 

18"""

73def _normalize_observables( 

74 observables: ObservableLike, 

75 n_qubits: int, 

76) -> tuple[list[Observable], bool]: 

77 """Return `(list_of_Observable, was_scalar)`. 

78 

79 `was_scalar` lets the outer layer squeeze a 1-element result back to a 

80 scalar. 

81 """ 

82 from .observables import PauliExpr # local import to avoid a cycle 

83 

84 if observables is None: 

85 return ([Observable.z(q) for q in range(n_qubits)], False) 

86 if isinstance(observables, PauliExpr): 

87 return ([observables.to_observable()], True) 

88 if isinstance(observables, Observable): 

89 return ([observables], True) 

90 result: list[Observable] = [] 

91 for obs in observables: 

92 if isinstance(obs, PauliExpr): 

93 result.append(obs.to_observable()) 

94 elif isinstance(obs, Observable): 

95 result.append(obs) 

96 else: 

97 raise TypeError(f"not an observable: {type(obs)}") 

98 return (result, False)

101def _reject_empty(observables: list[Observable]) -> None: 

102 """Refuse an observable with no Pauli terms. 

103 

104 An empty `Observable` is a legitimate value — `jordan_wigner` returns one 

105 with an offset when the operator reduces to a multiple of the identity — 

106 but it is not something a circuit can measure, and the backends do not 

107 agree on what to answer: three would return 0.0 and the rest would raise 

108 three different exception types. It is rejected once here rather than 

109 eight times, differently, further down. 

110 

111 Args: 

112 observables (list[Observable]): The normalized observables. 

113 

114 Raises: 

115 ValueError: If any of them carries no Pauli terms. 

116 """ 

117 for position, observable in enumerate(observables): 

118 if not observable.terms: 

119 where = f" at index {position}" if len(observables) > 1 else "" 

120 raise ValueError( 

121 f"observable{where} has no Pauli terms; there is nothing to " 

122 f"measure. A constant operator carries its value beside the " 

123 f"observable — the chemistry entry points return it as `offset`." 

124 )

127class Circuit: 

128 """A compiled circuit, ready to evaluate. 

129 

130 It holds a `CircuitSpec`, the observables to measure and a compiled backend 

131 handle. The circuit is compiled once, when the instance is created, and every 

132 call reuses the compiled handle. The instance does not store weights: you pass 

133 the `weights` at each call. 

134 

135 This is the NumPy interface. The top-level `Circuit(...)` function of the 

136 package returns this class for `interface="numpy"` and a `TorchCircuit` for 

137 `interface="torch"`. 

138 

139 Args: 

140 spec: The circuit to run, usually built with `assemble(blocks)`. 

141 observables: What to measure. `None` measures `Z` on every qubit and returns 

142 one value per qubit. A single `Observable` or `PauliExpr` returns a 

143 single value, and a list of them returns one value for each. Default: 

144 `None`. 

145 provider: The backend that runs the circuit. It can be a registered name 

146 (`"numpy"`, `"pennylane"` and others), a configuration instance, or a 

147 provider object. Default: `"numpy"`. 

148 interface: How a call is evaluated. With `"numpy"`, a call runs the 

149 NumPy path and returns NumPy values. With `"torch"`, each call is 

150 passed to the registered PyTorch interface, and `TorchCircuit` builds 

151 its core circuit in this way. The top-level `Circuit` function always 

152 passes `"numpy"`. Default: `"numpy"`. 

153 

154 Example: 

155 ```python 

156 import numpy as np 

157 from eigenkit import Circuit, Encoding, Variational, all_z 

158 

159 circuit = Circuit( 

160 blocks=[Encoding(in_features=2), Variational(depth=1)], 

161 observables=all_z(2), 

162 ) 

163 

164 # The circuit stores no weights, so you pass them at every call. 

165 weights = np.zeros(circuit.spec.n_weights) 

166 values = circuit(weights, np.array([0.1, 0.2])) 

167 

168 # Gradients, without PyTorch. 

169 grad = circuit.jacobian(weights, np.array([0.1, 0.2])) 

170 ``` 

171 """ 

172 

173 def __init__( 

174 self, 

175 spec: CircuitSpec, 

176 observables: ObservableLike = None, 

177 provider: ProviderSpec = "numpy", 

178 interface: Interface = "numpy", 

179 ): 

180 observable_list, scalar = _normalize_observables(observables, spec.n_qubits) 

181 _reject_empty(observable_list) 

182 self.spec = spec 

183 self.observables = observable_list 

184 self._scalar = scalar 

185 self.provider: Provider = resolve_provider(provider) 

186 self.interface = interface 

187 validate_circuit(spec, self.provider.capabilities) 

188 # Compile once; reuse for every call. 

189 self.compiled: CompiledCircuit = self.provider.compile(spec, observable_list) 

190 # Decide jacobian dispatch up front. Naive parameter-shift is only 

191 # correct when each gate angle is a direct WeightRef that no other 

192 # angle shares; anything wrapping a weight (Scale/Multiply/Sum/Negate, 

193 # e.g. `Reuploading(trainable_scale=True)`) or reusing one needs the 

194 # chain-rule path, which perturbs each parametric gate's angle 

195 # individually and weights it by the symbolic chain factor. 

196 self._has_unsafe_gates: bool = bool(find_parameter_shift_unsafe_gates(spec)) 

197 # Weights and inputs feeding a multi-frequency gate (a controlled 

198 # rotation): the two-term shift rule is not exact for those, so the 

199 # shift jacobians switch to the four-term rule for them alone. 

200 self._four_term_weights = weights_needing_four_term_shift(spec) 

201 self._four_term_inputs = inputs_needing_four_term_shift(spec) 

202 # Same check for inputs: when every input-dependent gate is a direct 

203 # InputRef used once, the naive input-shift jacobian is correct and 

204 # avoids `shift_gate_angle` (which wraps in `Sum`), so backends that 

205 # don't support `Sum` (notably cudaq) still get input grads. 

206 self._has_unsafe_input_gates: bool = bool(find_input_shift_unsafe_gates(spec)) 

207 # Cache of `(gate_index, shift_value)` -> angle-shifted compiled circuit, 

208 # lazily populated on the first backward pass through chain-rule. 

209 self._chain_rule_cache: dict[tuple[int, float], CompiledCircuit] = {}

214 if self.interface == "numpy": 

215 if inputs is None: 

216 inputs = self._default_inputs()

223 raise ValueError(f"unknown interface {self.interface!r}")

238 Args: 

239 given (int): Length the caller passed.

253 def _check_parameter_lengths(self, weights, inputs) -> None: 

254 """Reject a `weights` or `inputs` the circuit cannot consume. 

255 

256 Checking `ndim` alone is not enough. Without a length check, extra 

257 features would be read past silently -- a 5-feature input to a 

258 3-feature circuit would return the 3-feature answer with no indication 

259 -- and too few would surface as `IndexError: index 3 is out of 

260 bounds`, which names neither the circuit nor the expected size. 

261 

262 Lives here, not on the numpy path, because `__call__` dispatches to an 

263 interface that resolves its own default `inputs` first: the guard has 

264 to run after that, and every interface needs it, not just numpy. Reads 

265 `np.shape`, so a framework tensor is measured where it lies -- no 

266 conversion and no device transfer. 

267 

268 Args: 

269 weights (Any): Weight vector or batch; numpy or framework-native. 

270 inputs (Any): Input vector or batch, same. 

271 

272 Raises: 

273 ValueError: If either trailing axis is not the length the spec needs. 

274 """ 

275 self._check_length(np.shape(weights)[-1], self.spec.n_weights, "weights") 

276 self._check_length(np.shape(inputs)[-1], self.spec.n_inputs, "inputs")

280 def _run_numpy(self, weights, inputs) -> np.ndarray | float: 

281 weights_np = np.asarray(weights, dtype=np.float64) 

282 inputs_np = np.asarray(inputs, dtype=np.float64) 

283 if inputs_np.ndim not in (1, 2): 

284 raise ValueError( 

285 f"inputs must be 1D (single sample) or 2D (batch), got ndim={inputs_np.ndim}" 

286 ) 

287 self._check_parameter_lengths(weights_np, inputs_np) 

288 output = self.compiled.run(weights_np, inputs_np) 

289 if not self._scalar: 

290 return output 

291 if inputs_np.ndim == 1: 

292 return float(output[0]) 

293 return output[:, 0]

297 def _gradient_jacobians( 

298 self, 

299 weights_np: np.ndarray, 

300 inputs_np: np.ndarray, 

301 *, 

302 need_input_grad: bool = False, 

303 need_weight_grad: bool = True, 

304 ) -> tuple[np.ndarray | None, np.ndarray | None]: 

305 """Return `(weight_jacobian, input_jacobian_or_None)`. 

306 

307 `weight_jacobian` has shape `(n_observables, n_weights)`; 

308 `input_jacobian` has shape `(n_observables, n_inputs)` when 

309 `need_input_grad` is True, otherwise None. With `need_weight_grad` 

310 False (frozen weights in a torch backward) the parameter-shift paths 

311 skip the weight shifts and return None in its place. 

312 

313 Provider-native gradients come first: any backend whose compiled 

314 circuit exposes a ``jacobian`` (the numpy sim's adjoint, PennyLane's 

315 own diff engine, ...) uses it — computed however that backend's device 

316 config specifies. We only fall back to our backend-agnostic 

317 parameter-shift when the provider supplies none. We extend a backend's 

318 capabilities, never override them. 

319 

320 The parameter-shift fallback itself has two paths: the fast naive 

321 weight-only path when (a) only weight grads are needed and (b) every 

322 gate angle is either a direct WeightRef or independent of all weights; 

323 else the chain-rule path computes weight and input jacobians in a 

324 single pass over gates, reusing each perturbed-circuit forward. 

325 

326 For a 2-D ``inputs`` batch the jacobians gain a leading batch axis — 

327 ``(B, n_observables, n_weights)`` and ``(B, n_observables, n_inputs)``. 

328 

329 Every gradient path — `jacobian()`, the torch backward and the 

330 variational spine — comes through here, so this is where a backend that 

331 refuses gradients (``gradient_method="none"``) is held to it. 

332 """ 

333 if self.provider.gradient_method == "none": 

334 raise RuntimeError( 

335 f"backend '{self.provider.name}' has gradients disabled " 

336 f"(gradient_method='none'): a parameter-shift pass on hardware " 

337 f"costs 2 * n_weights paid jobs. Train on a simulator and deploy " 

338 f"the weights, or set enable_gradients=True in the device config " 

339 f"to opt in." 

340 ) 

341 if inputs_np.ndim == 2: 

342 return self._batched_gradient_jacobians( 

343 weights_np, 

344 inputs_np, 

345 need_input_grad=need_input_grad, 

346 need_weight_grad=need_weight_grad, 

347 ) 

348 provider_jacobian = getattr(self.compiled, "jacobian", None) 

349 if provider_jacobian is not None: 

350 return provider_jacobian( 

351 weights_np, inputs_np, need_input_grad=need_input_grad 

352 ) 

353 if not need_input_grad and not self._has_unsafe_gates: 

354 weight_jac = parameter_shift_jacobian( 

355 self.compiled, weights_np, inputs_np, self._four_term_weights 

356 ) 

357 return weight_jac, None 

358 if ( 

359 need_input_grad 

360 and not self._has_unsafe_gates 

361 and not self._has_unsafe_input_gates 

362 ): 

363 # Both jacobians via naive shifts. Avoids `shift_gate_angle` 

364 # so backends without `Sum` support (cudaq) still work. 

365 weight_jac = ( 

366 parameter_shift_jacobian( 

367 self.compiled, weights_np, inputs_np, self._four_term_weights 

368 ) 

369 if need_weight_grad 

370 else None 

371 ) 

372 input_jac = parameter_shift_input_jacobian( 

373 self.compiled, weights_np, inputs_np, self._four_term_inputs 

374 ) 

375 return weight_jac, input_jac 

376 return self._chain_rule_jacobians( 

377 weights_np, 

378 inputs_np, 

379 need_input_grad=need_input_grad, 

380 need_weight_grad=need_weight_grad, 

381 )

383 def _batched_gradient_jacobians( 

384 self, 

385 weights_np: np.ndarray, 

386 inputs_np: np.ndarray, 

387 *, 

388 need_input_grad: bool = False, 

389 need_weight_grad: bool = True, 

390 ) -> tuple[np.ndarray | None, np.ndarray | None]: 

391 """Jacobians for a 2-D ``inputs`` batch, with a leading batch axis. 

392 

393 A backend whose compiled circuit advertises a batch-capable ``jacobian`` 

394 (``supports_batched_jacobian``, e.g. the numpy sim's adjoint) computes 

395 the whole batch in one sweep; every other backend falls back to looping 

396 the single-sample path and stacking — same result, just without the 

397 per-sample vectorization. (Note: a backend can batch the *forward* pass 

398 natively yet still have a single-sample ``jacobian``, so this is a 

399 distinct, jacobian-specific flag — not ``capabilities.supports_batching``.) 

400 """ 

401 provider_jacobian = getattr(self.compiled, "jacobian", None) 

402 if provider_jacobian is not None and getattr( 

403 self.compiled, "supports_batched_jacobian", False 

404 ): 

405 return provider_jacobian( 

406 weights_np, inputs_np, need_input_grad=need_input_grad 

407 ) 

408 weight_jacs: list[np.ndarray | None] = [] 

409 input_jacs: list[np.ndarray | None] = [] 

410 for row in inputs_np: 

411 weight_jac, input_jac = self._gradient_jacobians( 

412 weights_np, 

413 row, 

414 need_input_grad=need_input_grad, 

415 need_weight_grad=need_weight_grad, 

416 ) 

417 weight_jacs.append(weight_jac) 

418 input_jacs.append(input_jac) 

419 batched_weight = ( 

420 None if weight_jacs[0] is None else np.stack(weight_jacs) 

421 ) # (B, n_obs, n_weights) 

422 batched_input = ( 

423 np.stack(input_jacs) if need_input_grad else None # (B, n_obs, n_inputs) 

424 ) 

425 return batched_weight, batched_input

427 def _chain_rule_jacobians( 

428 self, 

429 weights_np: np.ndarray, 

430 inputs_np: np.ndarray, 

431 *, 

432 need_input_grad: bool = False, 

433 need_weight_grad: bool = True, 

434 ) -> tuple[np.ndarray | None, np.ndarray | None]: 

435 """Chain-rule-aware parameter-shift jacobians. 

436 

437 For each parametric gate `g` with angle `theta_g = f(weights, inputs)`: 

438 

439 d<O>/dw_i = sum_g (dtheta_g/dw_i) * (d<O>/dtheta_g) 

440 d<O>/dx_j = sum_g (dtheta_g/dx_j) * (d<O>/dtheta_g) 

441 

442 `d<O>/dtheta_g` comes from parameter-shift on a circuit where `g`'s 

443 angle is wrapped in `Sum(original, Constant(shift))`: shifts of 

444 +/- pi/2, and also +/- 3*pi/2 for the gates in `MULTI_FREQUENCY_GATES` 

445 (`CRY`), whose four-term rule needs both. Perturbed compiled circuits 

446 are cached per `(gate_index, shift)`, so the cost is paid once across 

447 all backward calls on this instance. 

448 

449 `dtheta_g/dw_i` and `dtheta_g/dx_j` are computed symbolically via 

450 `chain_factor` / `chain_factor_input` at the current point. Both 

451 jacobians share the same `gate_angle_grad`, so adding the input 

452 jacobian costs only the extra chain-factor evaluations. 

453 """ 

454 n_obs = len(self.observables) 

455 n_weights = len(weights_np) 

456 weight_jac = ( 

457 np.zeros((n_obs, n_weights), dtype=np.float64) if need_weight_grad else None 

458 ) 

459 input_jac = ( 

460 np.zeros((n_obs, len(inputs_np)), dtype=np.float64) 

461 if need_input_grad 

462 else None 

463 ) 

464 

465 for gate_index, gate in enumerate(self.spec.gates): 

466 if not gate.params: 

467 continue 

468 angle = gate.params[0] 

469 weight_deps = ( 

470 collect_weight_dependencies(angle) if need_weight_grad else set() 

471 ) 

472 input_deps = collect_input_dependencies(angle) if need_input_grad else set() 

473 if not weight_deps and not input_deps: 

474 continue 

475 weight_factors: dict[int, float] = {} 

476 for weight_index in weight_deps: 

477 factor = chain_factor(angle, weight_index, weights_np, inputs_np) 

478 if factor != 0.0: 

479 weight_factors[weight_index] = float(factor) 

480 input_factors: dict[int, float] = {} 

481 for input_index in input_deps: 

482 factor = chain_factor_input(angle, input_index, weights_np, inputs_np) 

483 if factor != 0.0: 483 ↛ 481line 483 didn't jump to line 481 because the condition on line 483 was always true

484 input_factors[input_index] = float(factor) 

485 if not weight_factors and not input_factors: 

486 continue 

487 

488 # Same rule choice as the naive path, but per gate rather than per 

489 # weight: here the gate whose angle is being shifted is known, so 

490 # the four-term rule is used exactly for the multi-frequency ones. 

491 rule = FOUR_TERM_RULE if gate.name in MULTI_FREQUENCY_GATES else TWO_TERM_RULE 

492 gate_angle_grad = None # shape (n_obs,) 

493 for shift, coefficient in rule: 

494 term = coefficient * self._perturbed_compiled(gate_index, shift).run( 

495 weights_np, inputs_np 

496 ) 

497 gate_angle_grad = ( 

498 term if gate_angle_grad is None else gate_angle_grad + term 

499 ) 

500 

501 if weight_jac is not None: 

502 for weight_index, factor in weight_factors.items(): 

503 weight_jac[:, weight_index] += factor * gate_angle_grad 

504 if input_jac is not None: 

505 for input_index, factor in input_factors.items(): 

506 input_jac[:, input_index] += factor * gate_angle_grad 

507 

508 return weight_jac, input_jac

510 def _perturbed_compiled( 

511 self, 

512 gate_index: int, 

513 shift_value: float, 

514 ) -> CompiledCircuit: 

515 """Lazily build (and cache) a CompiledCircuit identical to this 

516 one except that one gate's angle is shifted by `shift_value`.""" 

517 key = (gate_index, shift_value) 

518 cached = self._chain_rule_cache.get(key) 

519 if cached is not None: 

520 return cached 

521 perturbed_spec = shift_gate_angle(self.spec, gate_index, shift_value) 

522 # A hardware handle compiles its shifted copies itself: on its own 

523 # device and qubits, so every gradient term comes from the same 

524 # hardware, and without repeating its cost warnings. 

525 compile_shifted = getattr(self.compiled, "compile_shifted", None) 

526 if compile_shifted is not None: 

527 compiled = compile_shifted(perturbed_spec) 

528 else: 

529 compiled = self.provider.compile(perturbed_spec, self.observables) 

530 self._chain_rule_cache[key] = compiled 

531 return compiled

535 def jacobian(self, weights, inputs=None, *, input_grad=False): 

536 r"""Return the gradient of the expectation values with respect to the weights. 

537 

538 It returns \(d\langle O \rangle / d(\mathrm{weights})\), of shape 

539 `(n_weights,)` for a single observable or `(n_observables, n_weights)` for a 

540 list of observables, like the output of `__call__()`. Pass a single 1-D 

541 `inputs` sample. For a batch, call it once for each row. 

542 

543 The NumPy interface can compute every gradient without PyTorch. This method 

544 uses the same sequence of gradient methods as the variational optimisation 

545 tools and the PyTorch interface. It uses the backend's own differentiation 

546 when the backend has one (the analytic adjoint method of the NumPy 

547 simulator, the differentiation of PennyLane, and so on), and otherwise the 

548 parameter-shift rule, which works on any backend. 

549 

550 With `input_grad=True`, it returns `(weight_jacobian, input_jacobian)`, 

551 where `input_jacobian` is \(d\langle O \rangle / d(\mathrm{inputs})\), with the 

552 same leading shape. 

553 

554 Args: 

555 weights (np.ndarray): The weight vector, of shape `(n_weights,)`. 

556 inputs (np.ndarray | None): One 1-D sample, of shape `(n_inputs,)`. This

641 def zero_state_probability(self, weights, inputs=None) -> float: 

642 r"""Return the probability of measuring the all-zeros state after the circuit. 

643 

644 It is useful for quantities based on fidelity: quantum kernels built from 

645 inversion circuits, swap tests and the Bures distance. Each backend 

646 computes it in its own way. PennyLane returns `qml.probs(...)[0]`, the 

647 Qiskit and NumPy simulators return the squared amplitude of the all-zeros 

648 basis state, Braket uses its own probability result, and IQM and Qiskit 

649 Runtime estimate it from shots. This method does not go through the 

650 `Observable` path, because the value is not the expectation value of a 

651 Pauli operator. 

652 

653 Args: 

654 weights (np.ndarray): The weight vector, of shape `(n_weights,)`. 

655 inputs (np.ndarray | None): One 1-D sample. `None` works only for a 

656 circuit that takes no inputs, and it raises a `ValueError` for any 

657 other circuit. Default: `None`. 

658 

659 Returns: 

660 The probability, a float in \([0, 1]\). 

661 

662 Raises: 

663 NotImplementedError: If the active backend does not provide 

664 `zero_state_probability`. 

665 """ 

666 probability = self._require_backend("zero_state_probability") 

667 if inputs is None: 

668 inputs = self._default_inputs() 

669 inputs_np = np.asarray(_to_numpy(inputs)) 

670 self._require_single_sample(inputs_np, "zero_state_probability") 

671 return float(probability(_to_numpy(weights), inputs_np))

707 def probabilities(self, weights, inputs=None) -> np.ndarray: 

708 """Return the measurement distribution over all `2**n_qubits` basis states. 

709 

710 The basis states are ordered with qubit `0` as the most significant bit. 

711 The distribution is used to decode samples. For example, the quantum 

712 approximate optimisation algorithm (QAOA) reads the most probable 

713 bitstrings from the optimised state. 

714 

715 Args: 

716 weights (np.ndarray): The weight vector, of shape `(n_weights,)`. 

717 inputs (np.ndarray | None): One 1-D sample. `None` works only for a 

718 circuit that takes no inputs, and it raises a `ValueError` for any 

719 other circuit. Default: `None`. 

720 

721 Returns: 

722 The probabilities of all basis states, of shape `(2**n_qubits,)`. 

723 

724 Raises: 

725 NotImplementedError: On a backend that has neither a `probabilities()` 

726 method nor a state vector to compute the probabilities from. 

727 """ 

728 # Derived from the statevector when the backend has no native 

729 # `probabilities`. Only four backends implement one, so without this 

730 # QAOA -- which reads its answer out of this -- would raise 

731 # NotImplementedError on qiskit, qiskit-aer, qulacs, qsim and braket, 

732 # four of which return a perfectly good statevector for the same 

733 # circuit. Deriving it here covers every backend that has one. 

734 native = getattr(self.compiled, "probabilities", None) 

735 if native is None and not hasattr(self.compiled, "statevector"): 

736 self._require_backend( 

737 "probabilities", "It has no statevector() to derive them from." 

738 ) 

739 if inputs is None: 

740 inputs = self._default_inputs() 

741 inputs_np = np.asarray(_to_numpy(inputs)) 

742 self._require_single_sample(inputs_np, "probabilities") 

743 if native is not None: 

744 return np.asarray(native(_to_numpy(weights), inputs_np), dtype=np.float64) 

745 return np.abs(np.asarray(self.statevector(weights, inputs_np))) ** 2

747 def sample(self, weights, inputs=None) -> dict[str, int]: 

748 """Return the raw measurement histogram, which maps each bitstring to a count.

764 other circuit. Default: `None`.

768 bitstring. The counts add up to the number of shots set for the 

769 backend.

773 every exact simulator, which has no shots to report. 

774 """

776 "sample",

785 def statevector(self, weights, inputs=None, initial_state=None) -> np.ndarray:

790 `(B, 2**n_qubits)`. With `initial_state`, a normalised flat statevector in

795 encoding.

800 only for a circuit that takes no inputs, and it raises a 

801 `ValueError` for any other circuit. Default: `None`. 

802 initial_state (np.ndarray | None): The state to start the evolution 

803 from, instead of \(|0 \cdots 0\rangle\). It is a normalised flat 

804 statevector in the same order. Default: `None`. 

805

807 The statevector, of shape `(2**n_qubits,)`, or `(B, 2**n_qubits)` for 

808 a batch.

812 which is the case for most quantum processing unit (QPU) and cloud 

813 backends.
