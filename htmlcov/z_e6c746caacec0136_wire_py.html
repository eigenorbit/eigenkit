1"""The JSON encoding of photonic circuits. 

2 

3This module is the photonic counterpart of `core.wire`. It turns a 

4`PhotonicCircuitSpec` into plain JSON data and back, so that a circuit can be sent 

5to a service or stored. It holds only data, and does not use Perceval. 

6 

7The angles of the components use the same encoding as the parameters of the qubit 

8circuits, so a `WeightRef` or an `InputRef` means the same thing in both. 

9"""

19def _plain(value): 

20 """Replace the NumPy scalars of an encoded angle with Python numbers. 

21 

22 The parameter encoding of the qubit circuits writes numbers as it finds them, 

23 and `json.dumps` refuses a NumPy integer or a 32-bit float. 

24 """ 

25 if isinstance(value, dict): 

26 return {key: _plain(item) for key, item in value.items()} 

27 if isinstance(value, list): 

28 return [_plain(item) for item in value] 

29 if isinstance(value, bool | str) or type(value) in (int, float): 

30 return value 

31 if isinstance(value, numbers.Integral): 

32 return int(value) 

33 if isinstance(value, numbers.Real): 33 ↛ 35line 33 didn't jump to line 35 because the condition on line 33 was always true

34 return float(value) 

35 return value

42def _component_to_json(comp) -> dict: 

43 if isinstance(comp, PhaseShifter): 

44 return {"c": "PS", "mode": int(comp.mode), "phi": _angle_to_json(comp.phi)} 

45 if isinstance(comp, BeamSplitter): 

46 return { 

47 "c": "BS", 

48 "modes": [int(m) for m in comp.modes], 

49 "theta": _angle_to_json(comp.theta), 

50 "phi": _angle_to_json(comp.phi), 

51 } 

52 if isinstance(comp, Permutation): 52 ↛ 54line 52 didn't jump to line 54 because the condition on line 52 was always true

53 return {"c": "PERM", "perm": [int(m) for m in comp.perm]} 

54 raise TypeError(f"unknown photonic component: {type(comp).__name__}")

57def _component_from_json(data: dict): 

58 tag = data["c"] 

59 if tag == "PS": 

60 return PhaseShifter(data["mode"], _param_from_json(data["phi"])) 

61 if tag == "BS": 

62 return BeamSplitter( 

63 tuple(data["modes"]), 

64 _param_from_json(data["theta"]), 

65 _param_from_json(data["phi"]), 

66 ) 

67 if tag == "PERM": 

68 return Permutation(tuple(data["perm"])) 

69 raise ValueError(f"unknown photonic component in JSON: {tag!r}")

72def photonic_spec_to_json(spec: PhotonicCircuitSpec) -> dict: 

73 """Encode a photonic circuit as JSON data. 

74 

75 Args: 

76 spec (PhotonicCircuitSpec): The circuit to encode. 

77 

78 Returns: 

79 A dictionary with the keys `"n_modes"`, `"n_weights"`, `"n_inputs"`, 

80 `"input_state"` and `"components"`. Every integer in it is a plain 

81 Python `int`, so `json.dumps` accepts it. 

82 """ 

83 return { 

84 "n_modes": int(spec.n_modes), 

85 "n_weights": int(spec.n_weights), 

86 "n_inputs": int(spec.n_inputs), 

87 "input_state": [int(n) for n in spec.input_state], 

88 "components": [_component_to_json(comp) for comp in spec.components], 

89 }
