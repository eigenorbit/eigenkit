1"""Quantum reservoir computing (QRC) estimators, compatible with scikit-learn. 

2 

3A reservoir is a circuit with fixed random weights, which are never trained. 

4These estimators sample the weights once (with the seed `random_state`), run the 

5circuit over the input sequences with 

6[`reservoir.run_reservoir()`][eigenkit.reservoir.run_reservoir], extract classical 

7features, and train only a linear readout. This split into a fixed reservoir and 

8a linear readout is the standard form of QRC. So these estimators belong with the 

9scikit-learn estimators (next to `QSVC` and `QSVR`), and not in the PyTorch 

10package. 

11 

12They are thin wrappers around the functions of the `reservoir` package. For full 

13control (custom processing of the features, custom readouts), call 

14`run_reservoir` and the feature extractors directly. 

15 

16Data shapes: `X` has shape `(n_sequences, n_timesteps, n_features)`, and `y` has 

17shape `(n_sequences, n_timesteps)`, with one target per timestep (sequence to 

18sequence). `washout` drops the first timesteps from the fit of the readout, so 

19that the reservoir can forget its initial state. Predictions are still made for 

20every timestep. 

21"""

62 """Shared reservoir forward pass for the regressor and classifier. 

63 

64 Subclasses provide the constructor attributes ``blocks``, ``state_policy``,

69 def _make_policy(self): 

70 if self.state_policy == "stateless": 

71 return Stateless(window=self.window) 

72 if self.state_policy == "carry": 

73 return Carry() 

74 if self.state_policy == "leaky": 

75 return LeakyCarry(alpha=self.leak) 

76 raise ValueError( 

77 f"state_policy must be 'stateless', 'carry', or 'leaky', " 

78 f"got {self.state_policy!r}" 

79 ) 

80 

81 def _build_reservoir(self) -> None: 

82 """Assemble the circuit and sample the frozen random reservoir weights.""" 

83 spec = assemble(self.blocks) 

84 self._circuit = Circuit(spec, all_z(spec.n_qubits), self.provider, "numpy") 

85 self._policy = self._make_policy() 

86 rng = np.random.default_rng(self.random_state) 

87 self._weights = rng.standard_normal(spec.n_weights) * self.weight_scale 

88 

89 def _features(self, X: np.ndarray) -> np.ndarray: 

90 """Drive the reservoir over ``X`` and extract features: ``(B, T, d)``.""" 

91 # `_circuit` is built by `fit`, so without this check reaching here first 

92 # would raise `AttributeError: no attribute '_circuit'` -- a traceback 

93 # into a private name rather than the NotFittedError every scikit-learn 

94 # estimator raises for this. 

95 check_is_fitted(self, "_circuit") 

96 traj = run_reservoir(self._circuit, self._policy, X, self._weights) 

97 spec = self.features 

98 if isinstance(spec, str) and spec in _NAMED_FEATURES: 

99 return _NAMED_FEATURES[spec](traj) 

100 return expectation_features(traj, spec) # Hamiltonian string(s) / Observable(s)

102 @staticmethod 

103 def _check_X(X: np.ndarray) -> np.ndarray: 

104 X = np.asarray(X, dtype=np.float64) 

105 if X.ndim != 3: 

106 raise ValueError( 

107 f"X must be 3-D (n_sequences, n_timesteps, n_features), got ndim={X.ndim}" 

108 ) 

109 return X

111 def _drop_washout(self, feats: np.ndarray, y: np.ndarray): 

112 """Flatten the post-washout (sequence, timestep) grid into rows. 

113 

114 Both validations live here, next to the slice they protect, so the two 

115 subclasses share one copy. Only `fit` reads `washout` -- 

116 prediction runs the whole sequence -- so this is every path a bad value 

117 can reach. 

118 """ 

119 B, T, d = feats.shape 

120 if self.washout < 0: 120 ↛ 123line 120 didn't jump to line 123 because the condition on line 120 was never true

121 # A negative value would index from the end, so `washout=-2` would 

122 # train on only the last two timesteps instead of dropping any. 

123 raise ValueError( 

124 f"washout must be >= 0; got {self.washout}. It is how many " 

125 f"leading timesteps to discard before fitting the readout." 

126 ) 

127 if self.washout >= T: 

128 raise ValueError(f"washout={self.washout} >= n_timesteps={T}") 

129 rows = feats[:, self.washout :, :].reshape(-1, d) 

130 targets = np.asarray(y)[:, self.washout :] 

131 return rows, targets

135class QuantumReservoirRegressor(_ReservoirMixin, BaseQuantumRegressor): 

136 r"""Quantum reservoir regressor: a fixed random reservoir and a ridge readout. 

137 

138 Args: 

139 blocks: The architecture of the reservoir circuit, a list of blocks that 

140 is assembled once. Its `n_inputs` must match the policy: `F` for the 

141 policies that carry the state, and `window * F` for 

142 `state_policy="stateless"`, where `F` is the number of features per 

143 timestep. 

144 state_policy: `"carry"` (the unitary of the circuit carries the state 

145 from one step to the next), `"stateless"` (each step starts from 

146 \(|0 \cdots 0\rangle\) and encodes a sliding window of inputs), or 

147 `"leaky"` (the carried state is mixed with a reset state and 

148 renormalised). 

149 window: The length of the sliding window, for `state_policy="stateless"`. 

150 leak: The mixing coefficient `alpha`, for `state_policy="leaky"`. 

151 features: The features read from the reservoir. It is `"z"` (the value 

152 of \(\langle Z \rangle\) on each qubit, the default), `"probabilities"`, 

153 `"amplitudes"`, a Pauli or Hamiltonian string (for example 

154 `"0.5 Z0 X1 - 0.3 Y0"`), or a list of such strings, to read out any 

155 observables. 

156 ridge: The strength of the L2 penalty of the linear readout, which is 

157 solved in closed form. 

158 washout: The number of timesteps dropped from the fit of the readout. 

159 They are not dropped from the predictions. 

160 weight_scale: The standard deviation of the sampled reservoir weights. 

161 provider: The backend, which must give access to the statevector. 

162 Default: `"numpy"`. 

163 random_state: The seed of the fixed reservoir weights. 

164 """

189 def fit(self, X: np.ndarray, y: np.ndarray) -> "QuantumReservoirRegressor": 

190 """Fit the estimator on `X` and `y`. 

191 

192 Args: 

193 X (np.ndarray): The sequences, of shape 

194 `(n_sequences, n_timesteps, n_features)`. A 2-D array is rejected. 

195 y (np.ndarray): One target for each timestep, of shape 

196 `(n_sequences, n_timesteps)`, or 

197 `(n_sequences, n_timesteps, n_outputs)` for several outputs. 

198 

199 Returns: 

200 This estimator, so that calls can be chained. 

201 """ 

202 X = self._check_X(X) 

203 self._build_reservoir() 

204 rows, targets = self._drop_washout( 

205 self._features(X), np.asarray(y, dtype=np.float64) 

206 ) 

207 targets = ( 

208 targets.reshape(-1) 

209 if targets.ndim == 2 

210 else targets.reshape(-1, targets.shape[-1]) 

211 ) 

212 self._readout = LinearReadout(ridge=self.ridge).fit(rows, targets) 

213 self.n_features_in_ = X.shape[-1] 

214 return self

221 """Predict a value for every timestep of every sequence in `X`. 

222 

223 Args: 

224 X (np.ndarray): The sequences, of shape 

225 `(n_sequences, n_timesteps, n_features)`. A 2-D array is rejected. 

226 

227 Returns: 

228 The predictions, of shape `(n_sequences, n_timesteps)`, or 

229 `(n_sequences, n_timesteps, n_outputs)` when the targets had

282class QuantumReservoirClassifier(_ReservoirMixin, BaseQuantumClassifier): 

283 r"""Quantum reservoir classifier: a fixed random reservoir and a logistic readout. 

284 

285 It uses the same reservoir as 

286 [`QuantumReservoirRegressor`][eigenkit.qml.qrc.QuantumReservoirRegressor]. 

287 The readout is a scikit-learn `LogisticRegression` on the features of each 

288 timestep. `y` holds integer or label targets, of shape 

289 `(n_sequences, n_timesteps)`. 

290 

291 Args: 

292 blocks: The architecture of the reservoir circuit, a list of blocks that 

293 is assembled once. Its `n_inputs` must match the policy: `F` for the 

294 policies that carry the state, and `window * F` for 

295 `state_policy="stateless"`, where `F` is the number of features per 

296 timestep. 

297 state_policy: `"carry"` (the unitary of the circuit carries the state 

298 from one step to the next), `"stateless"` (each step starts from 

299 \(|0 \cdots 0\rangle\) and encodes a sliding window of inputs), or 

300 `"leaky"` (the carried state is mixed with a reset state and 

301 renormalised). 

302 window: The length of the sliding window, for `state_policy="stateless"`. 

303 leak: The mixing coefficient `alpha`, for `state_policy="leaky"`. 

304 features: The features read from the reservoir. It is `"z"` (the value 

305 of \(\langle Z \rangle\) on each qubit, the default), `"probabilities"`, 

306 `"amplitudes"`, a Pauli or Hamiltonian string (for example 

307 `"0.5 Z0 X1 - 0.3 Y0"`), or a list of such strings, to read out any 

308 observables. 

309 C: The inverse regularisation strength of the logistic readout, passed 

310 to scikit-learn. Smaller values regularise more. This is the only 

311 argument that the regressor does not have: there, the readout is a 

312 ridge regression controlled by `ridge`. 

313 washout: The number of timesteps dropped from the fit of the readout. 

314 They are not dropped from the predictions. 

315 weight_scale: The standard deviation of the sampled reservoir weights. 

316 provider: The backend, which must give access to the statevector. 

317 Default: `"numpy"`. 

318 random_state: The seed of the fixed reservoir weights. 

319 """

344 def fit(self, X: np.ndarray, y: np.ndarray) -> "QuantumReservoirClassifier": 

345 """Fit the estimator on `X` and `y`. 

346 

347 Args: 

348 X (np.ndarray): The sequences, of shape 

349 `(n_sequences, n_timesteps, n_features)`. A 2-D array is rejected. 

350 y (np.ndarray): One target for each timestep, of shape 

351 `(n_sequences, n_timesteps)`. 

352 

353 Returns: 

354 This estimator, so that calls can be chained. 

355 """ 

356 from sklearn.linear_model import LogisticRegression 

357 from sklearn.preprocessing import LabelEncoder 

358 

359 X = self._check_X(X) 

360 self._build_reservoir() 

361 rows, targets = self._drop_washout(self._features(X), np.asarray(y)) 

362 self._label_encoder = LabelEncoder() 

363 encoded = self._label_encoder.fit_transform(targets.reshape(-1)) 

364 self._clf = LogisticRegression(C=self.C, max_iter=1000).fit(rows, encoded) 

365 self.classes_ = self._label_encoder.classes_ 

366 self.n_features_in_ = X.shape[-1] 

367 return self

369 def __sklearn_is_fitted__(self) -> bool: 

370 """Return True when the readout has been fitted.""" 

371 return hasattr(self, "_clf")

373 def predict(self, X: np.ndarray) -> np.ndarray: 

374 """Predict a label for every timestep of every sequence in `X`. 

375 

376 Args: 

377 X (np.ndarray): The sequences, of shape 

378 `(n_sequences, n_timesteps, n_features)`. A 2-D array is rejected. 

379 

380 Returns: 

381 The predicted labels, of shape `(n_sequences, n_timesteps)`. 

382 """ 

383 X = self._check_X(X) 

384 feats = self._features(X) 

385 B, T, d = feats.shape 

386 encoded = self._clf.predict(feats.reshape(B * T, d)) 

387 return self._label_encoder.inverse_transform(encoded).reshape(B, T)
