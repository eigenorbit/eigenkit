14class QSVC(BaseQuantumClassifier): 

15 """Quantum support vector classifier (QSVC). 

16 

17 It computes a quantum kernel matrix from a fidelity circuit, and then uses 

18 `SVC(kernel="precomputed")` from scikit-learn. 

19 

20 Args: 

21 feature_map: The feature map. It is `"Angle"` (angle encoding into a 

22 product state) or `"AngleCNOT"` (angle encoding 

23 with ladders of controlled-NOT (CNOT) gates). Default: `"Angle"`. It 

24 is ignored when `kernel` is given. 

25 reps: The number of repetitions of the feature map. Default: `1`. It is 

26 ignored when `kernel` is given. 

27 C: The regularisation parameter of the support vector machine. Default: 

28 `1.0`. 

29 probability: If True, the model gives Platt-scaled probabilities. If 

30 False (the default), `predict_proba` is unavailable. The calibration 

31 is a sigmoid fitted by cross-validation on the training kernel, as in 

32 `CalibratedClassifierCV(SVC(), ensemble=False)`. The predictions still 

33 come from the plain `SVC`. 

34 provider: The backend. It takes the same options as 

35 `Circuit(provider=...)`. Any backend with a statevector or a 

36 `zero_state_probability` method works, including Qiskit Runtime and 

37 IQM hardware. Default: `"numpy"`. It is ignored when `kernel` is given. 

38 random_state: The seed. 

39 kernel: A [`QuantumKernel`][eigenkit.qml.QuantumKernel] to use instead 

40 of building one from `feature_map`, `reps` and `provider`. It can be a 

41 custom kernel, a trained kernel or a hardware kernel. Default: `None`. 

42 scale: The bandwidth. The inputs are multiplied by it before they reach 

43 the kernel, so it can be tuned with `GridSearchCV`. Default: `1.0`. 

44 

45 Example: 

46 ```python 

47 import numpy as np 

48 from eigenkit import QSVC, Encoding, QuantumKernel, Variational 

49 

50 rng = np.random.default_rng(0) 

51 X_train, X_test = rng.uniform(0, np.pi, (8, 4)), rng.uniform(0, np.pi, (3, 4)) 

52 y_train = np.array([0, 1] * 4) 

53 

54 model = QSVC(feature_map="AngleCNOT", reps=2) 

55 model.fit(X_train, y_train) 

56 predictions = model.predict(X_test) 

57 

58 # Any kernel works, including custom and trained ones. 

59 my_map = [Encoding(in_features=4), Variational(depth=1)] 

60 model = QSVC(kernel=QuantumKernel(n_features=4, feature_map=my_map)) 

61 ``` 

62 """

113 def _new_svc(self) -> SVC: 

114 # `probability` is never passed: scikit-learn 1.9 deprecates it, even 

115 # as False. Probabilities come from a separate calibration instead. 

116 return SVC(kernel="precomputed", C=self.C, random_state=self.random_state)

124 def fit(self, X: np.ndarray, y: np.ndarray) -> "QSVC": 

125 """Fit the classifier on training data. 

126 

127 Args: 

128 X: The samples, of shape `(n_samples, n_features)`. 

129 y: The class labels, of shape `(n_samples,)`. 

130 

131 Returns: 

132 This estimator, so that calls can be chained. 

133 """ 

134 X = np.asarray(X, dtype=np.float64) 

135 self.kernel_ = _estimator_kernel(self, X.shape[1]) 

136 K_train = self.kernel_.compute_kernel_matrix(X * self.scale) 

137 svc = self._new_svc() 

138 svc.fit(K_train, y) 

139 self._svc = svc 

140 self._calibrated = None 

141 if self.probability: 

142 from sklearn.calibration import CalibratedClassifierCV 

143 

144 # Small QPU datasets can have fewer than 5 samples per class. 

145 cv = int(min(5, np.unique(y, return_counts=True)[1].min())) 

146 self._calibrated = CalibratedClassifierCV( 

147 self._new_svc(), method="sigmoid", cv=cv, ensemble=False 

148 ).fit(K_train, y) 

149 self._X_train = X 

150 self.classes_ = self._svc.classes_ 

151 self.n_features_in_ = X.shape[1] 

152 return self
